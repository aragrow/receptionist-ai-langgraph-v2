### Step 5: Update Workflow Graph

In `src/workflow/ai_receptionist_workflow.py`, add the new nodes:

```python
from src.nodes.clarification_handler import ClarificationHandler
from src.nodes.human_escalation import HumanEscalationNode

# Add nodes to workflow
workflow.add_node("clarification", clarification_handler)
workflow.add_node("human_escalation", human_escalation_node)

# Add conditional edges
workflow.add_conditional_edges(
    "l2_router",
    lambda state: "clarification" if state.missing_slots and state.clarification_count < state.max_clarifications else "l3_router",
    {
        "clarification": "clarification",
        "l3_router": "l3_router",
        "escalate": "human_escalation"
    }
)

workflow.add_conditional_edges(
    "clarification",
    lambda state: "l2_router" if not state.requires_human_escalation else "human_escalation",
    {
        "l2_router": "l2_router",
        "human_escalation": "human_escalation"
    }
)
```

async def test_full_clarification_flow():
    """Test complete clarification loop."""
    # 1. User request with missing info
    state = WorkflowState(
        speech_text="I want to schedule a cleaning",
        missing_slots=["preferred_date", "address"]
    )
    
    # 2. Generate clarification question
    state = await clarification_handler(state)
    assert state.awaiting_clarification is True
    
    # 3. User provides partial info
    state.speech_text = "October 15th"
    state = await clarification_handler(state)
    assert "preferred_date" in state.entities
    assert "address" in state.missing_slots
    
    # 4. Ask again for remaining info
    state = await clarification_handler(state)
    assert state.clarification_count == 2
    
    # 5. User provides remaining info
    state.speech_text = "123 Main Street, Minneapolis"
    state = await clarification_handler(state)
    assert len(state.missing_slots) == 0

async def test_escalation_after_failed_clarifications():
